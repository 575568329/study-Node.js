## 📁 fs文件系统

### 异步操作（推荐）

```javascript
const fs = require('fs').promises;

// 读取文件
const data = await fs.readFile('./file.txt', 'utf-8');

// 写入文件
await fs.writeFile('./file.txt', 'content', 'utf-8');

// 追加写入
await fs.appendFile('./file.txt', '\nmore content');

// 删除文件
await fs.unlink('./file.txt');

// 获取文件信息
const stats = await fs.stat('./file.txt');
console.log(stats.size);      // 文件大小
console.log(stats.mtime);     // 修改时间

// 读取目录
const files = await fs.readdir('./dir');
```

### 同步操作（谨慎使用）

```javascript
const fs = require('fs');

// 读取文件
const data = fs.readFileSync('./file.txt', 'utf-8');

// 写入文件
fs.writeFileSync('./file.txt', 'content', 'utf-8');
```

### 流操作

```javascript
const fs = require('fs');

// 创建读取流
const readStream = fs.createReadStream('./input.txt', {
  encoding: 'utf-8',
  highWaterMark: 64 * 1024
});

// 创建写入流
const writeStream = fs.createWriteStream('./output.txt');

// 管道连接
readStream.pipe(writeStream);

// 监听事件
readStream.on('data', (chunk) => console.log(chunk));
readStream.on('end', () => console.log('完成'));
readStream.on('error', (err) => console.error(err));
```

## 🌐 http服务器

### 创建服务器

```javascript
const http = require('http');

const server = http.createServer((req, res) => {
  // 获取请求信息
  console.log(req.method);  // GET, POST, etc.
  console.log(req.url);     // /path?query=value
  console.log(req.headers); // 请求头对象

  // 返回响应
  res.statusCode = 200;
  res.setHeader('Content-Type', 'text/html; charset=utf-8');
  res.end('<h1>Hello</h1>');
});

server.listen(3000, () => {
  console.log('服务器运行在 http://localhost:3000');
});
```

### 返回JSON

```javascript
const http = require('http');

const server = http.createServer((req, res) => {
  res.setHeader('Content-Type', 'application/json');

  const data = {
    code: 200,
    message: 'success',
    data: [{ id: 1, name: '张三' }]
  };

  res.end(JSON.stringify(data));
});

server.listen(3000);
```

### 处理不同请求方法

```javascript
const http = require('http');

const server = http.createServer((req, res) => {
  if (req.method === 'GET' && req.url === '/api/users') {
    // 获取用户列表
    res.setHeader('Content-Type', 'application/json');
    res.end(JSON.stringify([{ id: 1, name: '张三' }]));

  } else if (req.method === 'POST' && req.url === '/api/users') {
    // 创建用户
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', () => {
      const data = JSON.parse(body);
      res.statusCode = 201;
      res.end(JSON.stringify({ code: 201, data }));
    });

  } else {
    res.statusCode = 404;
    res.end('Not Found');
  }
});

server.listen(3000);
```

### 获取URL参数

```javascript
const http = require('http');
const url = require('url');

const server = http.createServer((req, res) => {
  const parsedUrl = url.parse(req.url, true);

  console.log(parsedUrl.pathname);   // /search
  console.log(parsedUrl.query);      // { keyword: 'nodejs', page: '1' }

  const keyword = parsedUrl.query.keyword;
  const page = parsedUrl.query.page;

  res.end(JSON.stringify({ keyword, page }));
});

server.listen(3000);
```

## ⏱️ 异步编程

### Promise

```javascript
// 创建Promise
const p = new Promise((resolve, reject) => {
  setTimeout(() => {
    resolve('成功');
    // 或
    reject('失败');
  }, 1000);
});

// 使用Promise
p.then(result => {
  console.log(result);
}).catch(err => {
  console.error(err);
});

// 链式调用
Promise.resolve(1)
  .then(x => x + 1)
  .then(x => x * 2)
  .then(x => console.log(x));  // 4
```

### async/await

```javascript
// async函数
async function test() {
  try {
    const data = await fs.readFile('./file.txt', 'utf-8');
    console.log(data);
    return data;
  } catch (err) {
    console.error('错误:', err);
    throw err;
  }
}

// 调用async函数
test().catch(err => console.error(err));

// 并行执行
const [data1, data2] = await Promise.all([
  fs.readFile('./file1.txt'),
  fs.readFile('./file2.txt')
]);
```

## 🚀 快速启动模板

### HTTP服务器模板

```javascript
const http = require('http');
const url = require('url');

const server = http.createServer(async (req, res) => {
  res.setHeader('Content-Type', 'application/json');

  if (req.method === 'GET' && req.url === '/api/data') {
    res.end(JSON.stringify({ code: 200, data: [] }));
  } else {
    res.statusCode = 404;
    res.end(JSON.stringify({ code: 404, message: 'Not Found' }));
  }
});

server.listen(3000, () => console.log('Server: http://localhost:3000'));
```

### 文件操作模板

```javascript
const fs = require('fs').promises;
const path = require('path');

async function processFile() {
  try {

    console.log('文件内容:', data);
  } catch (err) {
    console.error('处理失败:', err);
  }
}

processFile();
```
